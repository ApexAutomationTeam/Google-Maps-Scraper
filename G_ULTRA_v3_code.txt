  // ---------- 0) keyword — search box sabse sahi hai (URL peeche reh jata hai) ----------

  // waghera) ab BLOCK NAHI hote — website Google ke apne field se aati hai, aur
  // agar business ne wahi link diya hai to wo uska ASLI link hai, kachra nahi.
  const JUNK_HOSTS=['gstatic.com','ggpht.com','googleusercontent.com','googleapis.com','google-analytics.com',
    'googletagmanager.com','doubleclick.net','googleadservices.com','gvt1.com','gvt2.com','goo.gl','g.co',
    'schema.org','w3.org','example.com','wixstatic.com','squarespace-cdn.com','shopifycdn.com'];
  const hostOf = u => { try { return new URL(u).hostname.toLowerCase().replace(/^www\./,''); } catch(e){ return ''; } };
  const isHost = (h,d) => h===d || h.endsWith('.'+d);
  const socialOf = u => { const h=hostOf(u); for(const [lab,ds] of SOCIAL_DEFS) if(ds.some(d=>isHost(h,d))) return lab; return ''; };
  // koi bhi google.<kuch bhi> domain (google.com, google.co.uk, google.pk...)
  const isGoogle = h => /(^|\.)google(apis|usercontent|tagmanager|adservices)?\.[a-z.]{2,7}$/.test(h) || /(^|\.)google\./.test(h);
  // tasveer / file / script — website nahi
  const ASSET_RE = /\.(png|jpe?g|jpg|gif|webp|avif|svg|ico|bmp|tiff?|mp4|webm|mov|mp3|css|js|json|xml|woff2?|ttf|eot|pdf)(\?|#|$)/i;
  const ASSET_PATH = /\/(thumbnail|streetview|photo|image|img|static|assets|cdn-cgi)\b/i;
  const isJunk = u => {
    const h = hostOf(u);
    if (!h || h==='localhost' || /^\d+\.\d+\.\d+\.\d+$/.test(h)) return true;
    if (isGoogle(h)) return true;
    if (JUNK_HOSTS.some(d => isHost(h,d))) return true;
    if (ASSET_RE.test(u)) return true;
    let path=''; try { path = new URL(u).pathname; } catch(e){}
    if (ASSET_PATH.test(path)) return true;
    return false;
  };

  /* ============ FIELD ENGINE ============================================
     Ahem: ab poore data me regex se kuch NAHI dhoonda jata. Har cheez Google
     ke apne mukarrar khane se aati hai, aur likhne se pehle validator pass
     karti hai. Jo validator fail kare wo khali chhod di jati hai — galat
     value file me kabhi nahi jayegi.
        Naam      b[11]        Category  b[13][0]     Rating/Reviews b[4][7], b[4][8]
        Place ID  b[78]        Address   b[183][1]    Phone          b[178][0][0]
        Website   b[7][0]      Price     b[4][2]      Hours          b[203][1]
     =================================================================== */
  const gv = (o, ...idx) => { let c=o; for(const i of idx){ if(c==null) return null; c=c[i]; } return c; };
  const str = v => (typeof v==='string') ? v.trim().replace(/\s+/g,' ') : (typeof v==='number' ? String(v) : '');

  const V = {                                            // har column ka validator
    name:    v => v.length>0 && v.length<=160 && !/^[0-9][A-Za-z0-9_\-]{18,}$/.test(v),   // token nahi
    text:    v => v.length>0 && v.length<=120,
    phone:   v => /^\+?\d[\d\s\-().]{5,24}$/.test(v),
    rating:  v => /^[0-5](\.\d)?$/.test(v),
    reviews: v => /^\d{1,9}$/.test(v),
    zip:     v => /^[A-Z0-9][A-Z0-9\- ]{2,9}$/i.test(v),
    plus:    v => /^[23456789CFGHJMPQRVWX]{4}\+[23456789CFGHJMPQRVWX]{2,3}$/.test(v),
    hours:   v => /^(Open|Closed|Closes|Opens)\b/.test(v) && v.length<=45,
    price:   v => /\d/.test(v) && v.length<=30,
    url:     v => { try { const u=new URL(v);
                      return /^https?:$/.test(u.protocol) && /^[a-z0-9.-]+\.[a-z]{2,}$/i.test(u.hostname) && !isJunk(v);
                    } catch(e){ return false; } },
    street:  v => v.length<=200 && !/https?:\/\//.test(v) && !/^[0-9][A-Za-z0-9_\-]{18,}$/.test(v),
  };
  const QC = {};                                          // kya kya reject hua — aakhir me report
  const ok = (val, test, tag) => {
    const v = str(val);
    if (!v) return '';
    if (test(v)) return v;
    QC[tag] = (QC[tag]||0) + 1;                           // reject hua — khali jayega
    return '';
  };

  const tidyUrl = u => { try{ const w=new URL(u); return w.origin + (w.pathname==='/'?'':w.pathname.replace(/\/$/,'')); }catch(e){ return ''; } };
  // share/login/intent wale link kisi ka profile nahi hote
  const BAD_SOCIAL_PATH = /\/(sharer|share_channel|login|signup|intent|tr|plugins|dialog|help|policies|legal)\b/i;
  const socialLink = u => {
    if (!V.url(u)) return '';
    let p=''; try{ p=new URL(u).pathname; }catch(e){}
    if (BAD_SOCIAL_PATH.test(p)) return '';
    if (p==='' || p==='/') return '';                     // sirf homepage — kisi ka profile nahi
    return tidyUrl(u);
  };

  const nodeToRec = (b) => {
    if (!Array.isArray(b)) return null;
    const name = ok(gv(b,11), V.name, 'name');
    if (!name) return null;
    if (/\(Owner\)|^Beverage menu|^Menu$/.test(name)) return null;
    // har asli listing ka place_id hota hai — jiska na ho wo Google ka internal token hai
    const pid = str(gv(b,78));
    if (!/^ChI[A-Za-z0-9_\-]{10,}$/.test(pid)) return null;

    const a = addrOf(b);

    // ---- Website: Google ka apna khana b[7][0] ----
    let website = ok(gv(b,7,0), V.url, 'website');
    website = website ? tidyUrl(website) : '';

    // ---- Social: sirf social hosts, aur sirf profile-jaise link ----
    const soc = {}; SOCIAL_DEFS.forEach(([l]) => soc[l] = '');
    if (website && socialOf(website)) { soc[socialOf(website)] = website; website = ''; }  // site khud FB page ho
    const blob = JSON.stringify(b);
    const found = [...new Set((blob.match(/https?:\\?\/\\?\/[^"\\]{4,300}/g)||[]).map(u=>u.replace(/\\/g,'')))];
    found.forEach(u => { const lab = socialOf(u); if (!lab || soc[lab]) return; const link = socialLink(u); if (link) soc[lab] = link; });

    // ---- Phone: Google ka apna khana b[178][0][0] ----
    const rawPhone = ok(gv(b,178,0,0), V.phone, 'phone');

    // ---- Hours: sirf b[203] ke andar dhoondo, poore blob me nahi ----
    let hours = '';
    const hs = gv(b,203,1);
    if (Array.isArray(hs)) for (const x of hs) {
      const cand = Array.isArray(x) ? str(x[0]) : str(x);
      if (cand && V.hours(cand)) { hours = cand; break; }
    }

    // ---- Maps URL ----
    const mapsUrl = 'https://www.google.com/maps/place/?q=place_id:' + pid;

    return {
      base:{ Keyword:keyword, 'Business Name':name,
        Category: ok(gv(b,13,0), V.text, 'category'),
        Rating:   ok(gv(b,4,7),  V.rating,  'rating'),
        Reviews:  ok(gv(b,4,8),  V.reviews, 'reviews'),
        Street:   ok(a.street, V.street, 'street'),
        City:     ok(a.city,    V.text, 'city'),
        State:    ok(a.state,   V.text, 'state'),
        Zip:      ok(a.zip,     V.zip,  'zip'),
        Country:  ok(a.country, V.text, 'country'),
        Phone:    fmtPhone(rawPhone),
        'Phone (Original)': rawPhone,
        Website:  website },
      soc,
      tail:{ 'Plus Code': ok(a.plus, V.plus, 'plusCode'),
        Hours: hours,
        Price: ok(gv(b,4,2), V.price, 'price'),
        'Google Maps URL': mapsUrl,
        'Full Address': '' }
    };
  };

  const seen = new Set(); const rows = [];
  let pageRaw = 0;                       // is page pe Google ne kitne bheje (duplicate milaa kar)
  const scanData = (P) => {
    let added=0; pageRaw=0;
    if (!P) return 0;
    const scan=(v,d)=>{ if(d>10||!Array.isArray(v))return; v.forEach(x=>{
      if(Array.isArray(x)&&Array.isArray(x[1])&&typeof x[1][11]==='string'){ pageRaw++; const r=nodeToRec(x[1]); if(r){ const k=r.base['Business Name'].toLowerCase().replace(/[^a-z0-9]/g,''); if(k&&!seen.has(k)){seen.add(k); rows.push(r); added++;} } }
      else scan(x,d+1); }); };
    scan(P,0); return added;
  };

  // ---------- 2) page by page ----------
  // Ruko sirf tab jab Google BILKUL khali page de, ya lagataar 2 page pe koi
  // naya business na aaye. (Pehle ek hi duplicate-bhare page pe ruk jate the —
  // isi wajah se kabhi kam kabhi zyada leads aate the.)
  let dry = 0;
  for (let p=0; p<MAX_PAGES; p++) {
    const url = base + (p===0 ? '' : `!7i20!8i${p*20}`);
    let added = 0;
    try {
      // page 1 ka data upar hi mil chuka hai — dobara fetch mat karo
      const P = (p===0 && firstPage) ? firstPage
              : parseResp(await (await ORIG_FETCH(url,{headers:{'accept-language':'en'}})).text());
      added = scanData(P);
    } catch (e) { console.warn(`  page ${p+1} error:`, e.message); }
    console.log(`  page ${p+1}: +${added} naye  |  total: ${rows.length}`);
    if (p > 0) {
      if (pageRaw === 0) { console.log('  (list khatam)'); break; }
      dry = (added === 0) ? dry + 1 : 0;
      if (dry >= 2) { console.log('  (list khatam)'); break; }
    }
    await sleep(PAGE_DELAY);
  }
  if (!rows.length) { console.warn('Kuch nahi mila.'); return; }

  // ---------- 3) city/country ki chhoti galtiyan theek karo ----------
  const top = o => (Object.entries(o).sort((a,b)=>b[1]-a[1])[0]||[''])[0];
  const fC={}, fK={};
  rows.forEach(r=>{ if(r.base.City) fC[r.base.City]=(fC[r.base.City]||0)+1; if(r.base.Country) fK[r.base.Country]=(fK[r.base.Country]||0)+1; });
  const DOM = FORCE_CITY || top(fC), DOMC = top(fK);
  // City ki jagah dukaandaar ne galti se "Sector"/"coloney" jaisa likh diya ho
  const NOTCITY = /^(sector|sactor|secter|colony|coloney|colnoy|block|street|st|road|rd|town|chowk|market|phase|plot|shop|floor|near|opposite|main|gali|lane|avenue|ave|plaza|mall|centre|center)\b/i;
  rows.forEach(r=>{
    const c = r.base.City;
    if ((!c || c.length<4 || /^[IVXLC]+$/i.test(c) || /^\d+$/.test(c) || (NOTCITY.test(c) && fC[c]===1)) && DOM) {
      if (c && /^([IVXLC]+|\d+)$/i.test(c)) r.base.Street = (r.base.Street ? r.base.Street+' '+c : c);  // "III" wapas street me
      r.base.City = DOM;
    }
    if (!r.base.Country && DOMC) r.base.Country = DOMC;
    r.base.Street = tidy(r.base.Street).replace(/^[\s,]+|[\s,]+$/g,'');
    r.tail['Full Address'] = [r.base.Street, r.base.City, r.base.State, r.base.Zip, r.base.Country].filter(Boolean).join(', ');
  });

  console.log(`✅ TOTAL ${rows.length} leads | phone: ${rows.filter(r=>r.base.Phone).length} | website: ${rows.filter(r=>r.base.Website).length} | city: ${DOM}, ${DOMC}`);

  // ---------- QC: file likhne se pehle khud check karo ----------
  const bad = { website:0, mapsUrl:0, phone:0, city:0, junkName:0, streetHasUrl:0 };
  rows.forEach(r => {
    const w=r.base.Website, m=r.tail['Google Maps URL'], p=r.base.Phone;
    if (w && !V.url(w)) bad.website++;
    if (!/^https:\/\/www\.google\.com\/maps\/place\/\?q=place_id:ChI/.test(m)) bad.mapsUrl++;
    if (p && !/^\d{3}-\d{3}-\d{2,8}$/.test(p)) bad.phone++;
    if (!r.base.City) bad.city++;
    if (/^[0-9][A-Za-z0-9_\-]{18,}$/.test(r.base['Business Name'])) bad.junkName++;
    if (/https?:\/\//.test(r.base.Street)) bad.streetHasUrl++;
  });
  const badTotal = Object.values(bad).reduce((a,b)=>a+b,0);
  const rejected = Object.entries(QC).filter(([,n])=>n>0);
  console.log(badTotal===0 ? '🔍 QC pass — koi kharab value nahi mili.'
                           : '🔍 QC: ' + JSON.stringify(bad));
  if (rejected.length) console.log('   (validator ne reject kiye: ' + rejected.map(([k,n])=>k+' x'+n).join(', ') + ' — wo khane khali chhode gaye)');
  if (rows.length < 15) {
    console.warn(`⚠️ Sirf ${rows.length} leads mile. Agar is search me is se zyada hone chahiye the,`);
    console.warn(`   to ye link address bar me kholein aur code dobara paste karein:  ${canonical}`);
  }

  // ---------- 4) xlsx ----------
  const presentSocials = SOCIAL_DEFS.map(([l])=>l).filter(l => l==='Facebook' || rows.some(x=>x.soc[l]));
  const baseCols=['Keyword','Business Name','Category','Rating','Reviews','Street','City','State','Zip','Country','Phone','Phone (Original)','Website'];
  const tailCols=['Plus Code','Hours','Price','Google Maps URL','Full Address'];
  const headers=[...baseCols,...presentSocials,...tailCols];
  const urlCols=new Set(['Website','Google Maps URL',...presentSocials]);
  const aoa=[headers];
  rows.forEach(l => aoa.push(headers.map(h => baseCols.includes(h)?(l.base[h]??''):(tailCols.includes(h)?(l.tail[h]??''):(l.soc[h]||'')))));

  const fname='maps_leads_'+keyword.replace(/\s+/g,'_').replace(/[^\w\-]/g,'');
  const colWidths = headers.map(h => h==='Business Name'?30 : h==='Street'?45 : (h==='Website'||h==='Google Maps URL')?38
                                  : h==='Full Address'?55 : h==='Phone (Original)'?18 : 14);
  const bytes = makeXlsx(aoa, { urlCols, colWidths });
  const el = document.createElement('a');
  el.href = URL.createObjectURL(new Blob([bytes], {type:'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet'}));
  el.download = fname + '.xlsx';
  document.body.appendChild(el); el.click(); el.remove();
  setTimeout(()=>URL.revokeObjectURL(el.href), 5000);
  console.log(`🎉 DONE — ${rows.length} leads. Download: ${fname}.xlsx`);
})();
